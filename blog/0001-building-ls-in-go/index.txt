import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)
 
func main() {
	path := "."
	showAll := false
	longFormat := false
 
	for _, arg := range os.Args[1:] {
		if !strings.HasPrefix(arg, "-") {
			path = arg
			continue
		}
		for _, char := range arg[1:] {
			switch char {
			case 'a':
				showAll = true
			case 'l':
				longFormat = true
			default:
				fmt.Fprintf(os.Stderr, "usage: %s [-al] [path]\n", os.Args[0])
				os.Exit(1)
			}
		}
	}
 
	entries, err := os.ReadDir(path)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
 
	var names []string
	if showAll {
		names = append(names, ".", "..")
	}
	for _, entry := range entries {
		if !showAll && strings.HasPrefix(entry.Name(), ".") {
			continue
		}
		names = append(names, entry.Name())
	}
 
	for _, name := range names {
		if !longFormat {
			fmt.Println(name)
			continue
		}
		info, err := os.Lstat(filepath.Join(path, name))
		if err != nil {
			fmt.Fprintln(os.Stderr, err)
			continue
		}
		fmt.Printf("%s %8d %s %s\n",
			info.Mode(),
			info.Size(),
			info.ModTime().Format("Jan _2 15:04"),
			name,
		)
	}
